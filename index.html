import Payboom from "@payboom/node";

const pb = new Payboom(process.env.PAYBOOM_KEY);

const charge = await pb.charges.create({
  amount: 4990,            // 49,90
  currency: "EUR",
  source: "tok_visa_4242",
  customer: "cus_8f3aB2",
  capture: true,
  metadata: { order: "ord_001" }

console.log(charge.status); // "succeeded"
